cmake_minimum_required(VERSION 3.22)
project(ros2_tensorrt_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Threads REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Tensor.msg" "msg/InferenceResult.msg" DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)

# OFF builds messages, validation tests and Python tools only; never a fake GPU node.
option(BUILD_GPU_NODE "Build the real CUDA/TensorRT inference executable" ON)
if(BUILD_GPU_NODE)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(TENSORRT_ROOT "" CACHE PATH "TensorRT installation prefix")
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS ${TENSORRT_ROOT}/include
    /usr/include/x86_64-linux-gnu /usr/include/aarch64-linux-gnu)
  find_library(TENSORRT_LIBRARY nvinfer HINTS ${TENSORRT_ROOT}/lib ${TENSORRT_ROOT}/lib64)
  find_library(TENSORRT_PLUGIN_LIBRARY nvinfer_plugin HINTS ${TENSORRT_ROOT}/lib ${TENSORRT_ROOT}/lib64)
  if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY OR NOT TENSORRT_PLUGIN_LIBRARY)
    message(FATAL_ERROR "TensorRT 10 C++ SDK required (headers, nvinfer, nvinfer_plugin). Set TENSORRT_ROOT for a tar installation.")
  endif()
  add_library(trt_backend STATIC src/engine.cpp src/preprocess.cu)
  target_include_directories(trt_backend PUBLIC include ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(trt_backend PUBLIC CUDA::cudart ${TENSORRT_LIBRARY} ${TENSORRT_PLUGIN_LIBRARY})
  set_target_properties(trt_backend PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(trt_backend PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
  add_executable(inference_node src/inference_node.cpp)
  target_include_directories(inference_node PRIVATE include)
  ament_target_dependencies(inference_node rclcpp rcl_interfaces sensor_msgs std_msgs)
  target_link_libraries(inference_node trt_backend "${cpp_typesupport_target}" Threads::Threads)
  target_compile_options(inference_node PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS inference_node DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY launch config docs models DESTINATION share/${PROJECT_NAME})
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/publish_test_images.py scripts/benchmark.py scripts/make_test_model.py
  scripts/build_engine.sh scripts/verify_fixture.py scripts/cpu_reference_demo.py scripts/demo_dashboard.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_include_directories(test_core PRIVATE include)
  if(BUILD_GPU_NODE)
    ament_add_gtest(test_preprocess test/test_preprocess.cpp)
    target_link_libraries(test_preprocess trt_backend)
  else()
    # Compile the actual ROS C++ node even without GPU SDK; do not link a fake backend.
    add_library(node_compile_check OBJECT src/inference_node.cpp)
    target_include_directories(node_compile_check PRIVATE include)
    ament_target_dependencies(node_compile_check rclcpp rcl_interfaces sensor_msgs std_msgs)
    target_link_libraries(node_compile_check "${cpp_typesupport_target}")
    target_compile_options(node_compile_check PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
